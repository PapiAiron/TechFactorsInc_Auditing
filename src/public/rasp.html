GPIO — General Purpose I/O

MicroPython
GPIO basics
from machine import Pin
import time

# Output pin
led = Pin(25, Pin.OUT)
led.value(1)          # HIGH
led.value(0)          # LOW
led.toggle()          # flip state
led.on()              # HIGH shorthand
led.off()             # LOW shorthand

# Input pin — floating
btn = Pin(14, Pin.IN)
print(btn.value())    # 0 or 1

# Input with pull-up resistor
btn = Pin(14, Pin.IN, Pin.PULL_UP)

# Input with pull-down resistor
btn = Pin(14, Pin.IN, Pin.PULL_DOWN)

# Open-drain output
od = Pin(5, Pin.OPEN_DRAIN)
od.value(0)           # actively drive LOW
od.value(1)           # release (high-Z)

# Blink loop
while True:
    led.toggle()
    time.sleep_ms(500)
C/C++ SDK
GPIO in C
#include "pico/stdlib.h"

#define LED_PIN  25
#define BTN_PIN  14

int main() {
    stdio_init_all();

    // Configure output
    gpio_init(LED_PIN);
    gpio_set_dir(LED_PIN, GPIO_OUT);

    // Configure input with pull-up
    gpio_init(BTN_PIN);
    gpio_set_dir(BTN_PIN, GPIO_IN);
    gpio_pull_up(BTN_PIN);
    // gpio_pull_down(BTN_PIN);

        gpio_put(LED_PIN, 1);  // HIGH
        sleep_ms(500);
        gpio_put(LED_PIN, 0);  // LOW

        // Read pin
        bool state = gpio_get(BTN_PIN);
    }
}
ℹ️  All GPIO pins on Pico W are 3.3V logic. Do NOT connect 5V signals directly. The onboard LED (GP25) is controlled via the Wi-Fi chip on Pico W — use machine.Pin("LED") instead of Pin(25) in MicroPython.

PWM — Pulse Width Modulation

16 PWM channels across 8 slices (A/B). Every GPIO can be assigned PWM. Frequency = sys_clk / (wrap + 1) / divider.

MicroPython
PWM control
from machine import Pin, PWM

# Basic PWM on GP15
pwm = PWM(Pin(15))
pwm.freq(1000)          # 1 kHz
pwm.duty_u16(32768)     # 50% (0–65535)
pwm.duty_ns(500000)     # 500µs high time

# LED fade effect
import math
led = PWM(Pin(15)); led.freq(1000)
for i in range(360):
    v = int((math.sin(math.radians(i)) + 1) / 2 * 65535)
    led.duty_u16(v)

# Servo control (50Hz, 1ms–2ms pulse)
servo = PWM(Pin(0)); servo.freq(50)
# 0°  → duty ~1638  (1ms / 20ms * 65535)
# 90° → duty ~4915  (1.5ms)
# 180°→ duty ~8192  (2ms)
servo.duty_u16(4915)   # 90 degrees

# Stop PWM
pwm.deinit()
C/C++ SDK
PWM in C
#include "hardware/pwm.h"

#define PWM_PIN 15

// Get PWM slice and channel
uint slice = pwm_gpio_to_slice_num(PWM_PIN);
uint chan  = pwm_gpio_to_channel(PWM_PIN);

gpio_set_function(PWM_PIN, GPIO_FUNC_PWM);

// Set wrap (top value) = freq control
pwm_set_wrap(slice, 65535);
// Set clock divider (reduce freq)
pwm_set_clkdiv(slice, 4.0f);
// Set duty cycle on channel A/B
pwm_set_chan_level(slice, PWM_CHAN_A, 32768);
pwm_set_enabled(slice, true);

// Helper: set percentage duty
void set_duty(uint slice, uint chan, float pct) {
    pwm_set_chan_level(slice, chan,
        (uint16_t)(pct / 100.0f * 65535));
}

ADC — Analog-to-Digital Converter

3 external ADC pins (GP26=ADC0, GP27=ADC1, GP28=ADC2) + internal temperature sensor (ADC4). 12-bit resolution (0–4095). Reference = 3.3V.

MicroPython 
from machine import ADC, Pin

# Read analog pin GP26 (ADC0)
adc = ADC(Pin(26))
raw = adc.read_u16()       # 0–65535
voltage = raw * 3.3 / 65535
print(f"Voltage: {voltage:.2f}V")

# Alternative: ADC channel number
adc0 = ADC(0)   # GP26
adc1 = ADC(1)   # GP27
adc2 = ADC(2)   # GP28

# Internal temperature sensor (ADC4)
temp_sensor = ADC(4)
raw = temp_sensor.read_u16()
voltage = raw * (3.3 / 65535)
celsius = 27 - (voltage - 0.706) / 0.001721
print(f"Temp: {celsius:.1f}°C")

# Read potentiometer (0–100%)
pot = ADC(Pin(28))
percent = pot.read_u16() / 655
print(f"Position: {percent:.1f}%")
C/C++ SDK 
#include "hardware/adc.h"

adc_init();
adc_gpio_init(26);     // ADC0 on GP26
adc_select_input(0);   // channel 0

uint16_t raw = adc_read();  // 0–4095

// Temperature sensor (channel 4)
adc_set_temp_sensor_enabled(true);
adc_select_input(4);
float v = adc_read() * 3.3f / 4095.0f;
float temp = 27.0f - (v - 0.706f) / 0.001721f;

// Round-robin multi-channel reading

uint16_t ch0 = adc_read();
uint16_t ch1 = adc_read();
uint16_t ch2 = adc_read();
⚠️  ADC pins must not exceed 3.3V. Use a voltage divider for higher voltage signals. AGND (pin 33) is the analog ground — use this for noise-sensitive measurements.

I2C — Inter-Integrated Circuit

Two I2C buses (I2C0, I2C1). Standard 100 kHz, Fast 400 kHz. Requires pull-up resistors (4.7kΩ typical) on SDA & SCL lines.

MicroPython 
from machine import I2C, Pin

# I2C0: SDA=GP20, SCL=GP21
i2c = I2C(0, sda=Pin(20), scl=Pin(21), freq=400000)

# Scan for devices
devices = i2c.scan()
print([hex(d) for d in devices])

# Write bytes to address 0x48
i2c.writeto(0x48, b'\x00')
i2c.writeto(0x48, bytes([0x01, 0x02]))

# Read 2 bytes from address
data = i2c.readfrom(0x48, 2)

# Write then read (register read pattern)
i2c.writeto(0x48, bytes([0x00]))  # set register
data = i2c.readfrom(0x48, 2)     # read result

# writeto_mem / readfrom_mem
i2c.writeto_mem(0x48, 0x01, bytes([0xFF]))
val = i2c.readfrom_mem(0x48, 0x00, 2)

# SoftI2C (bit-bang any pins)
from machine import SoftI2C
si2c = SoftI2C(sda=Pin(4), scl=Pin(5))

SPI — Serial Peripheral Interface

Two SPI controllers (SPI0, SPI1). Up to 62.5 Mbps. Full-duplex. Pins: SCK (clock), TX/MOSI, RX/MISO, CS (chip select).

MicroPython 
from machine import SPI, Pin

# SPI0: SCK=GP18, MOSI=GP19, MISO=GP16
spi = SPI(0,
    baudrate=1_000_000,
    polarity=0,          # CPOL
    phase=0,             # CPHA
    bits=8,
    firstbit=SPI.MSB,
    sck=Pin(18),
    mosi=Pin(19),
    miso=Pin(16)
)
cs = Pin(17, Pin.OUT, value=1)  # CS high=deselect

# Transfer data
cs.low()
spi.write(b'\x01\x02')         # send bytes
data = spi.read(4)             # receive 4 bytes
cs.high()

# Full-duplex write+read simultaneously
cs.low()
rx = bytearray(4)
spi.write_readinto(b'\xAB\xCD', rx)
cs.high()

# SoftSPI (bit-bang any pins)
from machine import SoftSPI
sspi = SoftSPI(baudrate=100000,
    sck=Pin(2), mosi=Pin(3), miso=Pin(4))

UART — Serial Communication

Two UART controllers. Default UART0 on GP0 (TX) and GP1 (RX) at 115200 baud is used for USB serial console. Cross-wire TX→RX and RX→TX between devices.

MicroPython 
from machine import UART, Pin

# UART1: TX=GP4, RX=GP5
uart = UART(1, baudrate=9600, tx=Pin(4), rx=Pin(5))

# Full configuration
uart = UART(1,
    baudrate=115200,
    bits=8,
    parity=None,      # None, 0=even, 1=odd
    stop=1,
    tx=Pin(4),
    rx=Pin(5),
    timeout=1000      # ms read timeout
)

# Send data
uart.write('Hello!\r\n')
uart.write(b'\x01\x02\x03')

# Receive data
if uart.any():                    # bytes waiting?
    line = uart.readline()         # read until \n
    data = uart.read(10)          # read N bytes
    data = uart.read()             # read all waiting

# Non-blocking with timeout=0
uart.timeout = 0
ch = uart.read(1)  # None if nothing available

Interrupts & IRQs

MicroPython 
from machine import Pin
import micropython

# Button interrupt — rising edge
btn = Pin(14, Pin.IN, Pin.PULL_UP)

count = 0

# IRQ handler (keep SHORT & fast!)
def btn_handler(pin):
    global count
    count += 1

btn.irq(
    trigger=Pin.IRQ_FALLING,   # active LOW button
    handler=btn_handler
)

# Trigger options:
# Pin.IRQ_RISING   — LOW→HIGH
# Pin.IRQ_FALLING  — HIGH→LOW
# Pin.IRQ_RISING | Pin.IRQ_FALLING — both

# Disable interrupt
btn.irq(handler=None)

# Schedule safe code from IRQ
def safe_handler(pin):
    micropython.schedule(process_event, pin)

def process_event(pin):
    # safe to do more work here
    print("Pressed")
C/C++ SDK 
#include "hardware/gpio.h"

volatile bool pressed = false;

void gpio_callback(uint gpio, uint32_t events) {
    if (gpio == 14) {
        pressed = true;
    }
}

int main() {
    gpio_init(14);
    gpio_set_dir(14, GPIO_IN);
    gpio_pull_up(14);

    // Set up IRQ
    gpio_set_irq_enabled_with_callback(
        14,
        GPIO_IRQ_EDGE_FALL,  // FALL | RISE | LOW | HIGH
        true,
        &gpio_callback
    );

    while (true) {
        if (pressed) {
            pressed = false;
            // handle event
        }
    }
}
⛔  Keep ISR/IRQ handlers extremely short. Never do I2C, SPI, UART, or print() inside an IRQ handler. Use flags/queues and process in the main loop.

import bluetooth
from micropython import const

_IRQ_CENTRAL_CONNECT    = const(1)
_IRQ_CENTRAL_DISCONNECT = const(2)
_IRQ_GATTS_WRITE        = const(3)

ble = bluetooth.BLE()
ble.active(True)

# Define GATT service + characteristic
_SERVICE_UUID = bluetooth.UUID('6E400001-...')
_TX_UUID      = bluetooth.UUID('6E400003-...')
_TX_CHAR = (_TX_UUID,
    bluetooth.FLAG_NOTIFY | bluetooth.FLAG_READ,)
_SERVICE = (_SERVICE_UUID, (_TX_CHAR,),)
((tx,),) = ble.gatts_register_services((_SERVICE,))

conn_handle = None

def irq(event, data):
    global conn_handle
    if event == _IRQ_CENTRAL_CONNECT:
        conn_handle, _, _ = data
    elif event == _IRQ_CENTRAL_DISCONNECT:
        conn_handle = None
        advertise()  # restart advertising

ble.irq(irq)

def advertise(interval_us=500000):
    ble.gap_advertise(interval_us, adv_data=b'\x02\x01\x06')

def send(data):
    if conn_handle is not None:
        ble.gatts_notify(conn_handle, tx, data)

advertise()
send(b'Hello BLE')

HC-SR04 — Measure Distance with Sound Pulses

📌 Wiring Diagram — ⚠ ECHO needs voltage divider!

# ────────────────────────────────────────────────
# WHAT WE NEED
# ────────────────────────────────────────────────
from machine import Pin
import time

# ────────────────────────────────────────────────
# SET UP THE TWO PINS
# TRIG = we SEND a short pulse to start measurement
# ECHO = we READ how long the echo pulse lasts
# ────────────────────────────────────────────────
trig = Pin(3, Pin.OUT)     # GP3 sends trigger pulse
echo = Pin(2, Pin.IN)      # GP2 reads echo return

# ────────────────────────────────────────────────
# HOW IT WORKS (step by step):
# 1. We send a 10µs HIGH pulse on TRIG
# 2. The sensor fires 8 ultrasound bursts at 40kHz
# 3. The bursts bounce off an object
# 4. ECHO pin goes HIGH for the travel time
# 5. distance = travel time ÷ 58 (gives cm)
#    (Speed of sound = 340 m/s = 1cm per 29µs,
#     round trip = ×2, so 58µs per cm)
# ────────────────────────────────────────────────
def measure_cm():
    trig.low()               # make sure TRIG is LOW
    time.sleep_us(2)        # settle for 2µs
    trig.high()              # send the trigger pulse
    time.sleep_us(10)       # hold HIGH for 10µs
    trig.low()               # end the trigger pulse

    # Wait for ECHO to go HIGH (pulse starts)
    while echo.value() == 0:
        pass                 # keep waiting
    start = time.ticks_us()  # record start time

    # Wait for ECHO to go LOW (pulse ends)
    while echo.value() == 1:
        pass                 # keep waiting

    # How long was the pulse? (microseconds)
    elapsed = time.ticks_diff(time.ticks_us(), start)

    # Convert to centimetres
    return elapsed / 58.0

# ────────────────────────────────────────────────
# MEASURE EVERY 0.5 SECONDS
# ────────────────────────────────────────────────
while True:
    dist = measure_cm()
    print(f"Distance: {dist:.1f} cm")
    time.sleep_ms(500)

⚡ Resistor Quick Reference — When and Why

ComponentResistorWhy — plain English
LED220Ω – 1kΩWithout a resistor, too much current flows and the LED burns out instantly. Formula: R = (3.3V − 2V) ÷ 0.02A = 65Ω minimum → always use at least 220Ω to be safe.
Button pull-up10kΩKeeps the GPIO pin at a known HIGH state when the button is not pressed. Without it the pin "floats" and gives random 0s and 1s. Using Pin.PULL_UP in code enables this internally — no physical resistor needed.
I2C SDA & SCL4.7kΩI2C wires are "open-drain" — they can only be pulled LOW. A pull-up resistor to 3.3V is required to pull them HIGH again. Most ready-made breakout boards already include these on the PCB.
HC-SR04 ECHO2kΩ + 1kΩ dividerThe ECHO pin outputs 5V but the Pico W GPIO pins are rated for 3.3V max. Connecting 5V directly WILL damage the chip. Voltage divider: 5V × 1kΩ/(2kΩ+1kΩ) = 1.67V which is safe.
DHT data line4.7kΩ – 10kΩThe DHT one-wire protocol requires the data line to be pulled HIGH between measurements. Connect from DATA pin to 3.3V rail.
NeoPixel DIN300Ω – 500ΩAbsorbs voltage spikes caused by the capacitance in long LED strips that could damage the Pico's GPIO pin. Place close to the Pico end of the wire, not the strip end.

4×4 Matrix Keypad — 8 pins

4×4 Keypad — Press any of 16 buttons (1–9, 0, A–D, *, #)

📌 Wiring — 8 wires: 4 Rows (OUT) + 4 Cols (IN)
🐍 MicroPython — Every line explained
# ────────────────────────────────────────────────
# WHAT WE NEED
# ────────────────────────────────────────────────
from machine import Pin
import time

# ────────────────────────────────────────────────
# SET UP ROW PINS (we drive these HIGH one at a time)
# and COL PINS (we read these to detect presses)
# Rows = OUTPUT,  Cols = INPUT with PULL_DOWN
# PULL_DOWN keeps cols LOW until a key bridges
# the row (HIGH) to the column
# ────────────────────────────────────────────────
ROWS = [Pin(p, Pin.OUT) for p in [2,3,4,5]]
COLS = [Pin(p, Pin.IN, Pin.PULL_DOWN) for p in [6,7,8,9]]

# ────────────────────────────────────────────────
# KEY MAP — matches physical layout on keypad
# Row 0: [1, 2, 3, A]
# Row 1: [4, 5, 6, B]
# Row 2: [7, 8, 9, C]
# Row 3: [*, 0, #, D]
# ────────────────────────────────────────────────
KEYS = [
    ['1', '2', '3', 'A'],
    ['4', '5', '6', 'B'],
    ['7', '8', '9', 'C'],
    ['*', '0', '#', 'D'],
]

# ────────────────────────────────────────────────
# SCAN FUNCTION — checks every row + col combo
# ────────────────────────────────────────────────
def get_key():
    for r, row in enumerate(ROWS):  # r=0,1,2,3
        row.high()                   # drive this row HIGH
        for c, col in enumerate(COLS): # check each col
            if col.value() == 1:    # col went HIGH!
                row.low()            # reset row
                return KEYS[r][c]  # return the key
        row.low()                    # no key in row
    return None                      # no key pressed

# ────────────────────────────────────────────────
# WAIT until a key is pressed, then return it
# ────────────────────────────────────────────────
def wait_key():
    while True:
        k = get_key()
        if k:
            time.sleep_ms(80)  # debounce delay
            return k

# ────────────────────────────────────────────────
# COLLECT A 4-DIGIT PIN (e.g. access control)
# ────────────────────────────────────────────────
print("Enter 4-digit PIN:")
pin = ""
while len(pin) < 4:
    k = wait_key()
    if k in '0123456789':  # only digits
        pin += k
        print('*', end='')   # hide digits
print()                        # new line
print("PIN entered:", pin)

🐍 MicroPython — Every line explained
# ────────────────────────────────────────────────
# WHAT IS A STEPPER MOTOR?
# Unlike a servo (angle) or DC motor (spin freely),
# a stepper motor moves in precise small "steps".
# Each step = a tiny rotation (about 0.7°).
# 512 steps = one full revolution (28BYJ-48).
# ────────────────────────────────────────────────
from machine import Pin
import time

# 4 coil control pins
pins = [Pin(p, Pin.OUT) for p in [2, 3, 4, 5]]

# ────────────────────────────────────────────────
# HALF-STEP SEQUENCE — 8 steps per cycle
# Each row = which coils are ON (1) or OFF (0)
# We cycle through this list to rotate the motor
# ────────────────────────────────────────────────
SEQ = [
    [1,0,0,0],   # step 1
    [1,1,0,0],   # step 2
    [0,1,0,0],   # step 3
    [0,1,1,0],   # step 4
    [0,0,1,0],   # step 5
    [0,0,1,1],   # step 6
    [0,0,0,1],   # step 7
    [1,0,0,1],   # step 8
]

def rotate(steps, clockwise=True, delay_ms=2):
    # clockwise=True goes forward, False goes backward
    seq = SEQ if clockwise else SEQ[::-1]
    for _ in range(steps):        # repeat N steps
        for s in seq:              # each step = 1 cycle
            for i, p in enumerate(pins):
                p.value(s[i])    # set each coil
            time.sleep_ms(delay_ms)

# Rotate 1 full revolution clockwise
rotate(512, clockwise=True)
time.sleep(1)
# Rotate 1 full revolution counter-clockwise
rotate(512, clockwise=False)

import time

# ────────────────────────────────────────────────
# WHAT IS A JOYSTICK?
# Inside are 2 potentiometers (variable resistors)
# Tilting left/right changes VRx voltage (X axis)
# Tilting up/down  changes VRy voltage (Y axis)
# Pressing the stick clicks a button (SW)
# ────────────────────────────────────────────────
from machine import ADC, Pin
import time

# ADC reads 0–65535 (0V to 3.3V)
vrx = ADC(Pin(26))              # X axis → GP26
vry = ADC(Pin(27))              # Y axis → GP27
sw  = Pin(15, Pin.IN, Pin.PULL_UP) # click button

CENTER = 32768   # midpoint value (joystick centred)
DEAD   = 3000    # ignore tiny movements near centre

while True:
    x = vrx.read_u16() - CENTER  # -32768 to +32767
    y = vry.read_u16() - CENTER  # negative=up, pos=down
    click = not sw.value()       # True when pressed

    # Decide direction
    direction = "CENTRE"
    if   x >  DEAD:  direction = "RIGHT"
    elif x < -DEAD:  direction = "LEFT"
    elif y >  DEAD:  direction = "DOWN"
    elif y < -DEAD:  direction = "UP"

    # Print values
    click_str = " [CLICK!]" if click else ""
    print(f"{direction:8}  X:{x:+6}  Y:{y:+6}{click_str}")
    time.sleep_ms(100)          # 10 readings per second
